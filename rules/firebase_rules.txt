rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isAuth() {
      return request.auth != null;
    }
    function userDoc() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data;
    }
    function userExists() {
      return isAuth() && exists(/databases/$(database)/documents/users/$(request.auth.uid));
    }
    function isActive() {
      return userExists() && userDoc().activo == true;
    }
    function isStaff() {
      return isActive() && userDoc().role != "client";
    }
    function roleIs(roles) {
      return isActive() && userDoc().role in roles;
    }
    function isClient() {
      return isActive() && userDoc().role == "client";
    }
    function myClientId() {
      return userDoc().clienteId;
    }
    function myEmpresaId() {
      return userDoc().empresaId;
    }
    function clienteOwns(resource) {
      return isClient() && (
        resource.data.clienteId == myClientId() ||
        (myEmpresaId() != null && resource.data.empresaId == myEmpresaId()) ||
        request.auth.uid in resource.data.uidsAsociados
      );
    }

    // ── Validaciones de esquema ──────────────────────────────

    // CAMBIO 1: email ya no es obligatorio (GABI no lo recolecta)
    // Se quitó 'email' de hasAll y se eliminó la validación de tamaño de email
    function validLead(data) {
      return data.keys().hasAll(['nombre', 'servicio', 'createdAt'])
          && data.keys().hasOnly(['nombre', 'apellido', 'email', 'telefono',
                                  'empresa', 'servicio', 'descripcion', 'createdAt', 'origen'])
          && data.nombre is string && data.nombre.size() > 0 && data.nombre.size() < 120
          && data.servicio is string
          && data.createdAt == request.time;
    }

    function validUserUpdate(data) {
      return data.keys().hasOnly(['nombre', 'apellido', 'telefono', 'foto',
                                  'activo', 'role', 'clienteId', 'empresaId',
                                  'updatedAt', 'lastLogin']);
    }

    function validCotizacion(data) {
      return data.clienteId is string
          && data.estado in [
               'borrador', 'enviada', 'aceptada', 'rechazada',
               'vencida', 'pendiente', 'aprobada', 'en_revision'
             ]
          && data.keys().hasOnly([
               'clienteId', 'empresaId', 'nombreCliente', 'telefono',
               'clienteTipoIdentificacion', 'clienteIdentificacion',
               'clienteNit', 'clienteNumeroDocumento', 'mostrarDocumento',
               'notas', 'notasArray', 'ubicacion', 'tipo', 'tipoCalculo',
               'formaPago', 'planPagos', 'items',
               'subtotal', 'total', 'monto', 'moneda',
               'mostrarValorLetras', 'anexos',
               'numero', 'titulo', 'servicio', 'descripcion',
               'estado', 'validoHasta', 'fecha',
               'linkPublico', 'linkWhatsApp',
               'creadoPor', 'createdAt', 'updatedAt',
               'aprobadoPor', 'aprobadoEn', 'rechazadoPor', 'rechazadoEn',
               'comentarioAprobacion'
             ]);
    }

    function validProyecto(data) {
      return data.clienteId is string
          && data.estado in [
               'pendiente', 'en_ejecucion', 'finalizado', 'suspendido',
               'activo', 'pausado', 'completado', 'cancelado', 'planificacion'
             ]
          && data.keys().hasOnly([
               'clienteId', 'empresaId', 'empresaNombre', 'nombreCliente',
               'clienteNombre',
               'numero', 'nombre', 'descripcion', 'estado',
               'fechaInicio', 'fechaCierre', 'fechaFin',
               'presupuesto', 'moneda',
               'tecnico', 'responsable', 'usuariosAsociados', 'aprobadores',
               'uidsAsociados',
               'evidencias', 'documentos',
               'cotizacionId', 'totalCotizado', 'planPagos', 'cotizacionEstado',
               'creadoPor', 'creadoEn', 'createdAt', 'updatedAt',
               'etapa', 'avance', 'notas'
             ]);
    }

    function validDocumento(data) {
      return data.keys().hasOnly(['clienteId', 'empresaId', 'tipo', 'titulo',
                                  'url', 'estado', 'proyectoId', 'creadoPor',
                                  'createdAt', 'updatedAt', 'fechaRespuesta',
                                  'descripcion', 'version'])
          && data.clienteId is string
          && data.tipo in ['contrato', 'acta', 'factura', 'cotizacion', 'informe',
                           'permiso', 'garantia', 'otro']
          && data.estado in ['pendiente', 'revisado', 'aprobado', 'rechazado', 'firmado'];
    }

    // ── users ────────────────────────────────────────────────
    match /users/{uid} {
      allow get:    if isAuth() && (request.auth.uid == uid || roleIs(["admin"]));
      allow list:   if roleIs(["admin"]);
      allow create: if roleIs(["admin"]);
      allow update: if (roleIs(["admin"]) || (isAuth() && request.auth.uid == uid))
                    && validUserUpdate(request.resource.data);
      allow delete: if roleIs(["admin"]);
    }

    // ── usuarios (colección legacy) ──────────────────────────
    match /usuarios/{uid} {
      allow read:   if roleIs(["admin"]) || (isAuth() && request.auth.uid == uid);
      allow write:  if roleIs(["admin"]) || (isAuth() && request.auth.uid == uid);
    }

    // ── clientes ─────────────────────────────────────────────
    match /clientes/{id} {
      allow get:    if isStaff() || (isClient() && myClientId() == id);
      allow list:   if isStaff();
      allow create: if roleIs(["admin", "comercial"]);
      allow update: if roleIs(["admin", "comercial"]);
      allow delete: if roleIs(["admin"]);
    }

    // ── clientInvites ────────────────────────────────────────
    match /clientInvites/{code} {
      allow get:    if isAuth();
      allow list:   if roleIs(["admin", "comercial"]);
      allow create: if roleIs(["admin", "comercial"])
                    && request.resource.data.enabled == true
                    && request.resource.data.claimedByUid == null;
      allow update: if isAuth();
      allow delete: if roleIs(["admin"]);
    }

    // ── cotizaciones ─────────────────────────────────────────
    match /cotizaciones/{id} {
      allow get:    if isStaff() || clienteOwns(resource);
      allow list:   if isStaff() || isClient();
      allow create: if roleIs(["admin", "comercial"])
                    && validCotizacion(request.resource.data);
      allow update: if roleIs(["admin", "comercial", "tecnico"])
                    && validCotizacion(request.resource.data);
      allow delete: if roleIs(["admin"]);
    }

    // ── publicQuotes ─────────────────────────────────────────
    match /publicQuotes/{id} {
      allow get:    if resource.data.enabled == true;
      allow list:   if false;
      allow create: if roleIs(["admin", "comercial"]);
      allow update: if roleIs(["admin", "comercial"]);
      allow delete: if roleIs(["admin"]);
    }

    // ── proyectos ────────────────────────────────────────────
    match /proyectos/{id} {
      allow get:    if isStaff() || clienteOwns(resource);
      allow list:   if isStaff() || isClient();
      allow create: if roleIs(["admin", "comercial", "tecnico"])
                    && validProyecto(request.resource.data);
      allow update: if (roleIs(["admin","comercial","tecnico"]) || clienteOwns(resource))
                    && validProyecto(request.resource.data);
      allow delete: if roleIs(["admin"]);
    }

    // ── documentos ───────────────────────────────────────────
    match /documentos/{id} {
      allow get:    if isStaff() || clienteOwns(resource);
      allow list:   if isStaff() || isClient();
      allow create: if roleIs(["admin", "comercial", "tecnico"])
                    && validDocumento(request.resource.data);
      allow update: if roleIs(["admin", "comercial", "tecnico"])
                    && validDocumento(request.resource.data);
      allow delete: if roleIs(["admin"]);
    }

    // ── leads ────────────────────────────────────────────────
    match /leads/{id} {
      allow get:    if isStaff();
      allow list:   if isStaff();
      // CAMBIO 2: staff puede crear leads manuales sin pasar validLead
      // El validLead protege solo los creates del sitio público (sin auth)
      allow create: if isStaff() || validLead(request.resource.data);
      allow update: if isStaff();
      allow delete: if roleIs(["admin", "comercial"]);
    }

    // ── cuentas ──────────────────────────────────────────────
    match /cuentas/{id} {
      allow read: if roleIs(["admin", "contador", "finanzas"]);
      allow write: if roleIs(["admin", "contador", "finanzas"]);
    }

    // ── movimientos ──────────────────────────────────────────
    match /movimientos/{id} {
      allow read: if roleIs(["admin", "contador", "finanzas"]);
      allow write: if roleIs(["admin", "contador", "finanzas"]);
    }

    // ── cuentas bancarias ────────────────────────────────────
    match /cuentasBancarias/{id} {
      allow read: if roleIs(["admin", "contador", "finanzas"]);
      allow write: if roleIs(["admin", "contador", "finanzas"]);
    }

    // ── logs ─────────────────────────────────────────────────
    match /logs/{id} {
      allow read: if roleIs(["admin", "contador", "finanzas"]);
      allow write: if roleIs(["admin"]);
    }

    // ── Bloqueo total por defecto ────────────────────────────
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
